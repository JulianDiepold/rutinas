No cambia nada más. // Hace 1 llamada REST con todo lo necesario en vez de abrir la planilla y leerla en varias llamadas. function apiDisponible_() { return typeof Sheets !== 'undefined' && !!Sheets && !!Sheets.Spreadsheets; } // Borra de la caché lo que se haya guardado de esa pestaña (se llama después de cada escritura). function invalidar_(sh) { const ss = sh.getParent().getId(); CacheService.getScriptCache().removeAll(['r_' + ss + '_cur', 'r_' + ss + '_' + sh.getName()]); } // Pestañas semanales (S1 - ..., S2 - ...) ordenadas de la más vieja a la más nueva.
