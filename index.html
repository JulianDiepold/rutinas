// ===== CONFIGURACIÓN ===== // Planilla "admin" (la tuya) con la pestaña: // - "Alumnos": A = ID_CLIENTE, B = TOKEN, C = ID de la planilla de ese alumno // (La pestaña "Dudas" ya no se usa: las observaciones van a la planilla de cada alumno.) const ADMIN_ID = '1pU4D4VPU0qYhYsC5cgBAP1BGQM2e2Y2LYm_HymWhG4c'; const ID_PLANILLA_PRUEBA = '1WKha84cLk848E4mvvsUzMmBuQVepZhg6eoU-7Tk3kx0'; // solo para diagnostico() // ===== ENTRADA WEB (API JSON) ===== // La página vive en GitHub Pages y habla con este script por fetch. // GET ?action=rutina&id=...&t=...[&semana=NOMBRE_PESTAÑA] // -> { ok:true, data:{ semana, actual, semanas:[últimas 4, la más nueva primero], dias } } // POST { action:'serie'|'sueno'|'obs', id, t, ... } (Content-Type: text/plain) function doGet(e) { const p = (e && e.parameter) || {}; try { if (p.action === 'rutina') return json_({ ok: true, data: getRutina(p.id, p.t, p.semana) }); return json_({ ok: false, error: 'Acción inválida' }); } catch (err) { console.error(err); return json_({ ok: false, error: msgErr_(err) }); } } function doPost(e) { try { const b = JSON.parse(e.postData.contents); switch (b.action) { case 'serie': guardarSerie(b.id, b.t, b.d, b.e, b.s, b.reps, b.peso, b.rir); break; case 'sueno': guardarSueno(b.id, b.t, b.d, b.horas); break; case 'obs': guardarObs(b.id, b.t, b.d, b.e, b.texto); break; default: return json_({ ok: false, error: 'Acción inválida' }); } return json_({ ok: true }); } catch (err) { console.error(err); return json_({ ok: false, error: msgErr_(err) }); } } function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); } // Solo se muestra el motivo cuando es el link; el resto queda en el registro de ejecución. function msgErr_(err) { return String(err && err.message) === 'Link inválido' ? 'Link inválido' : 'Error del servidor'; } // ===== FUNCIONES DE DATOS ===== // Devuelve la semana pedida (por defecto la actual). Solo se pueden pedir las últimas 4 pestañas. // Las escrituras (guardarSerie, guardarSueno, guardarObs) siempre van a la semana actual. function getRutina(id, token, semana) { const hojas = hojasSemanas_(planilla_(id, token)); const ultimas = hojas.slice(-4); const actual = ultimas[ultimas.length - 1]; let sh = actual; if (semana) { sh = ultimas.find(h => h.getName() === String(semana)); if (!sh) throw new Error('Semana inválida'); } return { semana: sh.getName(), actual: sh.getName() === actual.getName(), semanas: ultimas.map(h => h.getName()).reverse(), dias: parsear_(sh) }; } function guardarSerie(id, token, d, e, s, reps, peso, rir) { const sh = semanaActual_(planilla_(id, token)); const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const dia = parsear_(sh)[d]; const fila = dia.ejercicios[e].series[s].fila; sh.getRange(fila, dia.al + 1, 1, 3).setValues([[n_(reps), n_(peso), n_(rir)]]); } finally { lock.releaseLock(); } return true; } function guardarSueno(id, token, d, horas) { const sh = semanaActual_(planilla_(id, token)); const dia = parsear_(sh)[d]; sh.getRange(dia.r0 + 3, dia.al + 1).setValue(n_(horas) + 'h'); // 1ª fila de las 4 combinadas return true; } // Guarda la observación de la alumna en la columna "Observaciones" de "Completar Alumno", // en la primera fila de las series de ese ejercicio. function guardarObs(id, token, d, e, texto) { const sh = semanaActual_(planilla_(id, token)); // Limita el largo y evita que un texto que empieza con = + - @ se interprete como fórmula const limpio = String(texto || '').slice(0, 1000).replace(/^[=+\-@]/, "'$&"); const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const dia = parsear_(sh)[d]; const fila = dia.ejercicios[e].series[0].fila; sh.getRange(fila, dia.al + 5).setValue(limpio); // al = REPs; +4 = Observaciones (base 0) → +5 en base 1 } finally { lock.releaseLock(); } return true; } // ===== AUXILIARES ===== // La lista de alumnos se guarda 10 min en caché para no abrir la planilla admin en cada pedido. // Si agregás o cambiás un alumno/token y necesitás que rija ya, ejecutá limpiarCache() a mano. function alumnos_() { const cache = CacheService.getScriptCache(); const hit = cache.get('alumnos'); if (hit) return JSON.parse(hit); const filas = SpreadsheetApp.openById(ADMIN_ID).getSheetByName('Alumnos').getDataRange().getValues() .map(r => [String(r[0]), String(r[1]), String(r[2])]); cache.put('alumnos', JSON.stringify(filas), 600); return filas; } function limpiarCache() { CacheService.getScriptCache().remove('alumnos'); } function planilla_(id, token) { const f = alumnos_().find(r => r[0] === String(id) && r[1] === String(token)); if (!f) throw new Error('Link inválido'); return SpreadsheetApp.openById(f[2]); } // Pestañas semanales (S1 - ..., S2 - ...) ordenadas de la más vieja a la más nueva. Nunca incluye "Datos". function hojasSemanas_(ss) { const hojas = ss.getSheets().filter(h => /^S\d+\s*-/.test(h.getName())); hojas.sort((a, b) => parseInt(a.getName().slice(1), 10) - parseInt(b.getName().slice(1), 10)); return hojas; } function semanaActual_(ss) { const hojas = hojasSemanas_(ss); return hojas[hojas.length - 1]; } function n_(x) { const n = parseFloat(String(x).replace(',', '.')); return isNaN(n) ? '' : n; } // Lee la pestaña semanal. Se guía por los textos de los encabezados (no por letras de columna), // y como en las celdas combinadas solo la de arriba a la izquierda trae valor, todo se lee desde ahí. function parsear_(sh) { const rango = sh.getDataRange(); const v = rango.getDisplayValues(); const txt = x => String(x).trim(); const buscar = (fila, re) => fila.findIndex(x => re.test(txt(x))); const inicios = []; v.forEach((fila, i) => { if (fila.some(x => /^D[ÍI]A\s*\d+/i.test(txt(x)))) inicios.push(i); }); // Los links de los videos se leen solo de las columnas "REFERENCIA" (rango angosto), // en vez de pedir el texto enriquecido y las fórmulas de toda la planilla. const vc = []; inicios.forEach(r0 => v[r0].forEach((x, k) => { if (/^REFERENCIA$/i.test(txt(x))) vc.push(k); })); let rt = [], fm = [], c0 = 0; if (vc.length) { c0 = Math.min.apply(null, vc); const rv = sh.getRange(1, c0 + 1, v.length, Math.max.apply(null, vc) - c0 + 1); rt = rv.getRichTextValues(); fm = rv.getFormulas(); } const enlace = (i, k) => { const t = (rt[i] || [])[k - c0]; let u = t && t.getLinkUrl(); if (!u && t) { const r = t.getRuns().find(x => x.getLinkUrl()); u = r && r.getLinkUrl(); } if (!u) { const m = /HYPERLINK\("([^"]+)"/i.exec(((fm[i] || [])[k - c0]) || ''); u = m && m[1]; } return u || ''; }; return inicios.map((r0, d) => { const h = v[r0], sub = v[r0 + 1]; const c = { ej: buscar(h, /^EJERCICIOS/i), ser: buscar(h, /^SERIES/i), rep: buscar(h, /^REPS?$/i), rir: buscar(h, /^RIR$/i), des: buscar(h, /^DESCANSO/i), obs: buscar(h, /^OBSERVACIONES/i) }; // En la fila de abajo hay dos grupos REPs/PESO/RIR/Descanso/Observaciones: // el 1º es "Completar Alumno", el 2º es la semana anterior (se lee por posición, no por título). const g = []; sub.forEach((x, k) => { if (/^REPS?$/i.test(txt(x))) g.push(k); }); const al = g[0], prev = g[1]; const vid = []; h.forEach((x, k) => { if (/^REFERENCIA$/i.test(txt(x))) vid.push(k); }); const fin = d + 1 < inicios.length ? inicios[d + 1] : v.length; const ejercicios = []; for (let i = r0 + 6; i < fin; i++) { // r0+2..r0+5 son las 4 filas del sueño if (!/^\d+$/.test(txt(v[i][c.ser]))) continue; // un ejercicio empieza donde hay nº de series const n = parseInt(v[i][c.ser], 10), series = [], obsAnt = []; for (let k = 0; k < n && v[i + k]; k++) { const f = v[i + k]; const o = txt(f[prev + 4] || ''); if (o) obsAnt.push(o); // observación que la alumna dejó la semana anterior series.push({ fila: i + 1 + k, ant: { reps: n_(f[prev]), peso: n_(f[prev + 1]), rir: n_(f[prev + 2]), des: txt(f[prev + 3] || '') }, hoy: { reps: n_(f[al]), peso: n_(f[al + 1]), rir: n_(f[al + 2]) } }); } ejercicios.push({ nombre: v[i][c.ej], reps: v[i][c.rep], rir: v[i][c.rir], descanso: v[i][c.des], nota: v[i][c.obs], obsHoy: txt(v[i][al + 4] || ''), antObs: obsAnt.join(' · '), videos: vid.map(k => enlace(i, k)).filter(Boolean), series }); } return { titulo: txt(h.find(x => /^D[ÍI]A\s*\d+/i.test(txt(x)))), r0, al, sueno: { ant: v[r0 + 2][prev], hoy: v[r0 + 2][al] }, ejercicios }; }); } // Ejecutalo a mano desde el editor para verificar que lee bien tu plantilla (ver "Registro de ejecución") function diagnostico() { const sh = semanaActual_(SpreadsheetApp.openById(ID_PLANILLA_PRUEBA)); Logger.log('Pestaña: ' + sh.getName()); parsear_(sh).forEach(d => { Logger.log(d.titulo + ' → ' + d.ejercicios.length + ' ejercicios, sueño previo: ' + d.sueno.ant); d.ejercicios.forEach(e => Logger.log(' ' + e.nombre + ' | ' + e.series.length + ' series | videos: ' + e.videos.length)); }); }
